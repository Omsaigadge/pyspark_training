-- CREATING ACCOUNT MASTER
create table acct_mstr(accno number(4) PRIMARY KEY, accname varchar2(30),
balance number(9,2))

insert into acct_mstr values(&accno,'&accname',&bal)

commit;

select * from acct_mstr;


-- CREATING TXN MASTER
create table txn_mstr(accno number(4) references acct_mstr, txn_type varchar2(1), amount number(9,2))


insert into txn_mstr values(&accno,'&txn_type',&amount)


-- cursor for batch job to update data daily in account master table
declare
    cursor trans_curs
    is
    select * from txn_mstr for update;
    begin
        for trans_rec in trans_curs
        loop
        if trans_rec.txn_type ='C' then
            update acct_mstr
            set balance=balance-trans_rec.amount
            where accno=trans_rec.accno;
        elsif trans_rec.txn_type='C' then
            update acct_mstr
            set balance=balance+trans_rec.amount
            where accno = trans_rec.accno;
            end if;
            delete from txn_mstr where current of trans_curs;
            end loop;
            end;
            
            
			
Ref cursor -- strong ref cursor (fixed return valuue)
		   -- weak ref cursor (no fixed value of return)
		   
		   
Strong REF CURSOR in Oracle is a cursor variable that has a predefined structure and fixed return type specified during its declaration. Because its shape is known at compile time,
----------------- the PL/SQL engine enforces type-safety. You can only open a strong REF CURSOR for queries that exactly match its declared return structure, reducing runtime errors		   

In Oracle PL/SQL, a weak REF CURSOR is a specific type of cursor variable that acts as a pointer to a result set but is declared without a fixed return structure.

-- employees working for that department

create or replace Function disp_emp(v_dept_id employees.department_id%type)
return sys_refcursor
is
emplist sys_refcursor;

begin
    open emplist for select last_name from employees
    where department_id=v_dept_id;
    return emplist;
    end;
    
    declare
        employee_list sys_refcursor;
        lname employees.last_name%type;
        begin
            employee_list:= disp_emp(60);
            loop
                fetch employee_list into lname;
                exit when employee_list%notfound;
                dbms_output.put_line(lname);
            end loop;
        end;

empno number(4), ename varchar2(20), sal number(9,2)
);

create or replace type emp_nst_ty as table of emp_obj;

--3. Create table emp_dept with the following Structure deptno number(4),dname varchar2(20),loc varchar2(20), emp_details emp_nst_ty
CREATE TABLE emp_dept (
    deptno      NUMBER(4),
    dname       VARCHAR2(20),
    loc         VARCHAR2(20),
    emp_details emp_nst_ty
)
NESTED TABLE emp_details STORE AS emp_details_tab;

--12.Write a plsql code to display the department details


SET SERVEROUTPUT ON;

BEGIN
    FOR d IN (SELECT * FROM emp_dept)
    LOOP
        DBMS_OUTPUT.PUT_LINE('Department Number: ' || d.deptno);
        DBMS_OUTPUT.PUT_LINE('Department Name: ' || d.dname);
        DBMS_OUTPUT.PUT_LINE('Location: ' || d.loc);

        DBMS_OUTPUT.PUT_LINE('Employee Details:');

        FOR i IN 1 .. d.emp_details.COUNT
        LOOP
            DBMS_OUTPUT.PUT_LINE(
                d.emp_details(i).empno || '  ' ||
                d.emp_details(i).ename || '  ' ||
                d.emp_details(i).sal
            );
        END LOOP;

        DBMS_OUTPUT.PUT_LINE('--------------------------');
    END LOOP;
END;

--14.Write a plsql code to Insert a new employee record for deptno 10



SET SERVEROUTPUT ON;

BEGIN
    UPDATE emp_dept
    SET emp_details = emp_details MULTISET UNION
        emp_nst_ty(
            emp_obj(103, 'Martin', 2800)
        )
    WHERE deptno = 10;

    IF SQL%ROWCOUNT > 0 THEN
        DBMS_OUTPUT.PUT_LINE(
            'Employee inserted successfully.'
        );
    ELSE
        DBMS_OUTPUT.PUT_LINE(
            'Department 10 not found.'
        );
    END IF;

    COMMIT;
END;
